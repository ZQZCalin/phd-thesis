\section{Proofs in Section \ref{sec:DP-O2NC_privacy}}

\subsection{Proof of Theorem \ref{thm:DP-O2NC:tree}}
\label{app:DP-O2NC:tree}

Before we prove the theorem, we first prove a more general composition of RDP mechanisms. For two datasets $Z,Z'\in\calZ$, we denote $Z\simeq_q Z'$ if $Z,Z'$ are neighbors and they differ at the $q$-th entry.

\begin{lemma}
\label{lem-aux:DP-O2NC:RDP-composition}
For any domain $\calZ$, let $\calR^{(i)}:\calS^{(1)}\times\dotsm\times\calS^{(i-1)}\times\calZ \to \calS^{(i)}$ be a sequence of algorithms such that $\calR^{(i)}(\vecx_{1:i-1},\cdot)$ is $(\alpha,\epsilon_i)$-RDP for all auxiliary inputs $\vecx_{1:i-1}\in \calS^{(i)}\times\dotsm\times\calS^{(i-1)}$. Let $f_{\calR^{(i)}(\vecx_{1:i-1}, Z)}$ be the distribution of $\calR^{(i)}(\vecx_{1:i-1}, Z)$. Suppose each dataset $Z\in\calZ$ has $m$ data, and for each $q\in[m]$, let
\begin{equation*}
    \textsc{In}(q) := \{ i\in[n] : f_{\calR^{(i)}(\cdot, Z)} = f_{\calR^{(i)}(\cdot, Z')}, \, \forall Z\simeq_q Z' \}, \quad \textsc{Out}(q) := [n]\setminus \textsc{In}(q).
\end{equation*} 
Let $\calA_n:\calZ\to\calS^{(1)}\times\dotsm\times\calS^{(n)}$ be the algorithm that given a dataset $Z\in\calZ$ sequentially computes $\vecx_i = \calR^{(i)}(\vecx_{1:i-1},z_i)$ for $i\in[n]$ and then outputs $\vecx_{1:n}$. Then $\calA_n$ is $(\alpha, \epsilon)$-RDP, where
\begin{equation*}
    \textstyle
    \epsilon \le \max_{q\in[m]} \sum_{i\in \textsc{Out}(q)} \epsilon_i.
\end{equation*}
\end{lemma}

\begin{proof}
Suppose $Z\simeq_q Z'$ and let $P,Q$ be the distributions of $\calA_n(Z), \calA_n(Z')$ respectively. 
% Since $\calA_n$ computes $\vecx_{1:n}$ sequentially, we can write 
Note that $P=P_1\times\dotsm\times P_n$ where $P_i$ is the distribution of $\calA_i(Z)$ given $\calA_{i-1}(Z)$, i.e., $P_i(\vecx_i|\vecx_{1:i-1}) = f_{\calR^{(i)}(\vecx_{1:i-1},Z)}(\vecx_i)$. Similarly, $Q=Q_1\times\dotsm\times Q_n$ where $Q_i(\vecx_i|\vecx_{1:i-1})=f_{\calR^{(i)}(\vecx_{1:i-1},Z')}(\vecx_i)$.

For all $i\in \textsc{In}(q)$, $P_i(\cdot|\vecx_{1:i-1})=Q_i(\cdot|\vecx_{1:i-1})$ for all $\vecx_{1:i-1}$ by definition of $\textsc{In}(q)$; and for all $i\in\textsc{Out}(q)$, since $\calR^{(i)}(\vecx_{1:i-1},\cdot)$ is $(\alpha,\epsilon_i)$-RDP, $D_\alpha(P_i(\cdot|\vecx_{1:i-1})\|Q_i(\cdot|\vecx_{1:i-1})\le \epsilon_i$. Therefore,
% for all $\vecx_{1:i-1}$. Therefore,
\begin{align*}
    &\int P_i(\vecx_i|\vecx_{1:i-1})^\alpha Q_i(\vecx_i|\vecx_{1:i-1})^{1-\alpha} \, d\vecx_i \\
    &= \exp({(\alpha-1)D_\alpha(P_i(\cdot|\vecx_{1:i-1})\|Q_i(\cdot|\vecx_{1:i-1})})
    \le \left\{
    \begin{array}{ll}
        1, & i\in \textsc{In}(q),  \\
        e^{(\alpha-1)\epsilon_i}, & i\in \textsc{Out}(q).
    \end{array}\right.
\end{align*}
We then conclude the proof by recursively integrating from $d\vecx_n$ to $d\vecx_1$:
\begin{equation*}
    e^{(\alpha-1)D_\alpha(P\|Q)} = \int \prod P_i(\vecx_i|\vecx_{1:i-1})^\alpha Q_i(\vecx_i|\vecx_{1:i-1})^{1-\alpha} \, d\vecx_n \dotsm d\vecx_1 \le \prod_{i\in\textsc{Out}(q)} e^{(\alpha-1)\epsilon_i}.
    \qedhere
\end{equation*}
\end{proof}

Next, we introduce several tree notations. Let $k\in\NN_0$ and consider a complete binary tree with $n=2^k$ leaves. Leaves are indexed by $i\in[n]$ (or $(i,i)$ interchangeably) in an ascending order from left to right, and every other node is indexed by $(i,j)$ where $i$ is its left-most child leaf and $j$ is its right-most child leaf. Let $\sT$ be the set of all nodes, 
% $\sT_{(i,j)}$ be the index of $(i,j)$ in $\sT$, 
and let $<$ be a total order on $\sT$ such that $(a,b) < (c,d)$ if either $b < d$ or $b=d, a<c$. In other words, we compare rightmost child first and break tie with leftmost child. We assume $(\sT,<)$ is sorted in ascending order, then it's valid to label a sequence w.r.t. $\sT$ (e.g., we can rewrite $a_1,\ldots,a_{|\sT|}$ as $a_{(i,j)}$ for $(i,j)\in\sT$). 
We denote $a_{(c,d)<(a,b)}$ as a short-hand notation of $\{a_{(c,d)}:(c,d)<(a,b)\}$.
Now we are ready to prove the theorem.
% Consequently, we can assume that $\sT_{(a,b)}<\sT_{(c,d)}$ iff $(a,b)<(c,d)$, i.e., $\sT$ is sorted in an ascending order.

% \begin{observation}
% Given input $r$, the \textsc{Node} function in Algorithm \ref{alg:DP-O2NC:tree} starts from the root $(1,n)$ and 
% \end{observation}


% \TreeMechanism*

\begin{proof}
% [Proof of Theorem \ref{thm:DP-O2NC:tree}]
For any $n\in\NN$, define $\sT$ as he tree set with $2^{\lceil\log_2 n\rceil}$ leaves.
Let $\calZ = \calZ^{(1:n)}$ and $S^{(a,b)}=\calX$ for all $(a,b)\in\sT$. For each $(a,b)\in \sT$, define $\calR^{(a,b)}: ( \prod_{(c,d)<(a,b)} \calS^{(c,d)} ) \times \calZ \to \calS^{(a,b)}$ as
\begin{equation}
    \calR^{(a,b)}(\vecy_{(c,d)<(a,b)},Z_{1:n}) = \sum_{i=a}^b \calM^{(i)}(\vecx_{1:i-1},Z_i)
    + \xi_b, 
    % \quad \tilde\calR^{(a,b)} = \calR^{(a,b)} + \xi_b,
    \label{eq:DP-O2NC:1-tree-1}
\end{equation}
where $\vecx_j = \sum_{(c,d)\in \textsc{Node}(j)} \vecy_{(c,d)}$ for each $j\in[i-1]$ and $\xi_b\sim N(0,\sigma_b^2I)$. We can check that $\vecx_j$ is well-defined: for all $j \le b$ and for all $(c,d)\in \textsc{Node}(j)$, we have $(c,d)<(a,b)$ by definition of $\textsc{Node}$ and thus $\vecx_{1:i-1}$ can be computed from $\vecy_{(i,j)<(a,b)}$ for all $i\in[a,b]$.

Next, since for all neighboring $Z_{1:n},Z_{1:n}'\in\calZ$, $\|\calM^{(i)}(\vecx_{1:i-1},Z_i)-\calM^{(i)}(\vecx_{1:i-1},Z_i')\| \le s_i$, $\calR^{(a,b)}(\vecy_{(c,d)<(a,b)},\cdot)$ has sensitivity bounded by $s_{(a,b)} := \max_{i\in[a,b]} s_i$. Therefore, upon adding Gaussian noise $\calN(0,\sigma_b^2I)$, $\calR^{(a,b)}(\vecy_{(c,d)<(a,b)},\cdot)$ is $(\alpha, \epsilon_{(a,b)})$-RDP where $\epsilon_{(a,b)} = {\alpha s_{(a,b)}^2}/{2\sigma_b^2}$.
% Also recall that adding Gaussian noise $N(0,\sigma^2I)$ to an algorithm with $\ell_2$-sensitivity $\gamma$ makes it $(\alpha, \alpha\gamma^2/2\sigma^2)$-RDP. Therefore, $\tilde\calR^{\sT_{(a,b)}}$ is $(\alpha,\epsilon_{\sT_{(a,b)}})$-RDP where $\epsilon_{\sT_{(a,b)}} = \alpha\sup_{i\in[a,b]} \gamma_i^2/2\sigma_b^2$. 

Now we are ready to apply Lemma \ref{lem-aux:DP-O2NC:RDP-composition}.
Let $\calA_\calR:\calZ\to\prod\calS^{(a,b)}$ be the composition of $\calR$, i.e., given dataset $Z_{1:n}$ sequentially computes $\vecy_{(a,b)}=\calR^{(a,b)}(\vecy_{(c,d)<(a,b)},Z_{1:n})$ and outputs $\vecy_{(a,b)\in\sT}$. Then by Lemma \ref{lem-aux:DP-O2NC:RDP-composition}, $\calA_\calR$ is $(\alpha,\epsilon)$-RDP where
\begin{equation*}
    \epsilon \le \max_{q} \sum_{(a,b) \in \textsc{Out}(q)} \epsilon_{(a,b)} = \max_{q} \sum_{(a,b)\in \textsc{Out}(q)} \max_{i\in[a.b]} \frac{\alpha s_i^2}{2\sigma_b^2}.
\end{equation*}
% Suppose $Z_{1:n}\simeq_q Z'_{1:n}$ and let $I(q)\in [n]$ be the index such that subset $Z_{I(q)}$ that contains the $q$-th data.
% By construction \eqref{eq:1-tree-1}, $\calR^{\sT_{(a,b)}}$ only depends on subsets $Z_i$ for $i\in [a,b]$, so $\calR^{\sT_{(a,b)}}(\cdot, Z_{1:n}) \neq \calR^{\sT_{(a,b)}}(\cdot, Z'_{1:n})$ if and only if $I(q)\in [a,b]$. Therefore, $\textsc{Out}(q) = \{\sT_{(a,b)}:I(q)\in[a,b]\}$, i.e., it corresponds to the set of nodes in the binary tree such that which have $I(q)$ as a child leaf. Since there is exactly one such node in each layer, $|\textsc{Out}(q)| = k+1$. Hence,
% \begin{equation*}
%     \epsilon \le \sup_{b\in[n], i\le b} \frac{(k+1)\alpha\gamma_i^2}{2\sigma_b^2}.
% \end{equation*}
% For all $q$ such that $Z_{1:n}\simeq_q Z'_{1:n}$, there are exactly $k+1$ such nodes $(a,b)\in\sT$ that $q\in[a,b]$ (i.e., one node in each layer of the tree, and there are $k+1$ layers). 
% Observe that $\textsc{Out}(q)=\{(a,b)\in\sT:q\in[a,b]\}$ by construction of $\calR^{(a,b)}$ in \eqref{eq:1-tree-1}. 
Observe that node $(a,b)\in\textsc{Out}(q)$ if and only if the $q$-th data in $Z_{1:n}$ is in $Z_i$ and $i\in[a,b]$.
Since there is exactly one such node in each of $\lceil\log_2 n\rceil+1$ layers in the binary tree associated with $\sT$, we have $|\textsc{Out}(q)|=\lceil\log_2 n\rceil+1 \le 2\ln n$. Consequently,
\begin{equation*}
    \epsilon = \frac{\alpha\rho^2}{2} \le \alpha\ln n \cdot \max_{b\in[n], i\le b}\frac{s_i^2}{\sigma_b^2}.
\end{equation*}
% Next, we convert RDP to DP guarantees. Recall that if $\calA$ is $(\alpha,\alpha\rho^2/2)$-RDP for all $\alpha>1$, then it's also $(\epsilon', \delta)$-DP for all $\delta\in(0,1)$ and $\epsilon'=\rho\sqrt{\log 1/\delta}$. Hence, upon substituting $\rho$, we have
% \begin{equation*}
%     \epsilon' \le  \sqrt{\log_2(2n)\log1/\delta} \sup_{b\in[n], i\le b} \frac{\gamma_i}{\sigma_b}.
% \end{equation*}
Finally, we conclude the proof by claiming that algorithm $\calA$ defined in the theorem can be derived from $\calA_\calR$ after post-processing. Observe that $\sum_{(a,b)\in \textsc{Node}(i)}\sum_{j=a}^b = \sum_{j=1}^i$, so by \eqref{eq:DP-O2NC:1-tree-1},
\begin{align*}
    \vecx_i 
    = \sum_{(a,b)\in\textsc{Node}(i)} \vecy_{(a,b)} 
    &= \sum_{(a,b)\in\textsc{Node}(i)} \left( \sum_{j=a}^b\calM^{(j)}(\vecx_{1:j-1}, Z_j) + \xi_b \right) \\
    &= \sum_{j=1}^i \calM^{(j)}(\vecx_{1:j-1},Z_j) + \textsc{Tree}(i).
\end{align*}
% Recall that $\textsc{Tree}(i) = \sum_{(a,b)\in\textsc{Node}(i)} \xi_b$, so $\vecy_{1:n}$ is the composition $\calA$ defined in the theorem.
Therefore, $\calA$ is indeed a post-processing of $\calA_\calR$, which concludes the proof.
\end{proof}


\subsection{Proof of Corollary \ref{cor:DP-O2NC:privacy}}

% \TreeCorollary*

\begin{proof}
Since the private oracle defined in Algorithm \ref{alg:DP-O2NC:oracle-variance-reduced} uses disjoint data in different iteration $k\in[K]$, Algorithm \ref{alg:DP-O2NC:PONC} is a post-processing of $\{(\tilde\vecg_1^k,\ldots,\tilde\vecg_T^k)\}_{k\in[K]}$. Therefore, it suffices to prove that for each $k\in[K]$, the composition $(\tilde\vecg_1^k,\ldots,\tilde \vecg_T^k)$ is $(\alpha,\alpha\rho^2/2)$-RDP.

The key is to observe that $(\tilde\vecg_1^k,\ldots,\tilde\vecg_T^k)$ can be written as the adaptive composition in Theorem \ref{thm:DP-O2NC:tree}. Let $\vecg_1^k = \calM^{(1)}(Z_1^k) \triangleq \grad_{f,\delta}(\vecw_1^k,Z_1^k)$ and $\vecd_i^k = \calM^{(i)}(\tilde\vecg_{1:i-1}^k, Z_i^k) \triangleq \diff_{f,\delta}(\vecw_i^k,\vecw_{i-1}^k,Z_i^k)$. This is well-defined because 
% $\vecw_1^k$ is initialized by the OCO algorithm and is thus data-independent, and 
$\vecw_i^k$ is a post-processing of $\tilde\vecg_{1:i-1}^k$ by construction of O2NC. Therefore, $\vecg_t^k=\vecg_1^k+\sum_{i=2}^t \vecd_i^k = \sum_{i=1}^t \calM^{(i)}(\tilde\vecg_{1:i-1}^k,Z_i^k)$, and Theorem \ref{thm:DP-O2NC:tree} can be applied.

By Lemma \ref{lem:DP-O2NC:grad-estimator}, the sensitivity of $\calM^{(1)}$ is bounded by $s_1 = \frac{2dL}{B_1}$. By Remark \ref{rmk:DP-O2NC:w-distance} and Lemma \ref{lem:DP-O2NC:diff-estimator}, the sensitivity of $\calM^{(i)}(\tilde\vecg_{1:i-1}^k,\cdot)$ is bounded by $s_i=\frac{2dL}{B_2\delta}\|\vecw_i^k-\vecw_{i-1}^k\| \le \frac{4dL}{B_2T}$ for all $i\ge 2$.
% because $\|\vecw_i^k-\vecw_{i-1}^k\| \le 2D = \frac{2\delta}{T}$. 
Since we assume $B_1\ge TB_2/2$, it holds that $s_1\le s_t$ for all $t\ge 2$. Therefore, upon setting $\sigma_t = {\sqrt{2\ln T}s_2}/{\rho}$ for all $t\in[T]$, Theorem \ref{thm:DP-O2NC:tree} implies that $(\tilde\vecg_1^k,\ldots\tilde\vecg_T^k)$ is $(\alpha,\alpha\rho^{\prime 2}/2)$-RDP where
\begin{equation*}
    % \textstyle
    \rho' \le \sqrt{2\ln T} \cdot \max_{b\in[n], i\le n} \frac{s_i}{\sigma_b} \le \sqrt{2\ln T} \cdot \frac{s_2}{\sigma_2} = \rho.
    \qedhere
\end{equation*}
\end{proof}


\subsection{Proof of Corollary \ref{cor:DP-O2NC:tree-noise}}

% \TreeNoise*

\begin{proof}
Recall that $\tree(t)=\sum_{(\cdot,i)\in\node(t)} \xi_i$. Since $\xi_i$'s are independent and $\xi_i\sim \calN(0,\sigma_i^2I)$,
\begin{equation*}
    \E\|\tree(t)\|^2
    = \sum_{(\cdot,i)\in\node(t)} \E\|\xi_i\|^2
    = \sum_{(\cdot,i)\in\node(t)} d\sigma_i^2
    \le 2\ln(t) d\sigma^2.
\end{equation*}
The last inequality follows from Remark \ref{rmk:DP-O2NC:tree} that $|\node(t)|\le 2\ln(t)$. We then conclude the proof by substituting the value of $\sigma$ defined in Corollary \ref{cor:DP-O2NC:privacy}.
\end{proof}
