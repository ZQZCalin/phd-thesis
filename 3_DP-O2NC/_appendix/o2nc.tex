% \BaseLemma*

\begin{proof}
First, by fundamental theorem of calculus, we have
\begin{align*}
    F(\vecx_{t+1}^k) - F(\vecx_t^k)
    &= \int_0^1 \langle \nabla F(\vecx_t^k + t(\vecx_{t+1}^k-\vecx_t^k)), \vecx_{t+1}^k-\vecx_t^k\rangle \, dt \\
    &= \int_0^1 \langle \nabla F(\vecx_t^k+t\Delta_t^k), \Delta_t^k\rangle \, dt
    = \Ex_{s_t^k} \langle \nabla F(\vecw_t^k), \Delta_t^k\rangle. 
\end{align*}
Next, upon taking the telescopic sum (and recall $x_1^{k+1} = x_{T+1}^k$), we have
\begin{align*}
    \E F(\vecx_{T+1}^K) - F(\vecx_1^1)
    &= \sum_{k=1}^K\sum_{t=1}^T \E[ F(\vecx_{t+1}^k) - F(\vecx_t^k) ] 
    = \sum_{k=1}^K\sum_{t=1}^T \E\langle \nabla F(\vecw_t^k), \Delta_t^k \rangle.
\end{align*}
Note that $\langle \nabla F(\vecw_t^k), \Delta_t^k \rangle = \langle \nabla F(\vecw_t^k), \vecu^k \rangle + \langle \tilde\vecg_t^k, \Delta_t^k-\vecu^k \rangle + \langle \nabla F(\vecw_t^k) - \tilde \vecg_t^k, \Delta_t^k - \vecu^k \rangle$. Therefore, by definition of $\vecu^k$ and $\regret_T(\vecu^k)$,
\begin{equation*}
    \sum_{t=1}^T\langle \nabla F(\vecw_t^k), \Delta_t^k \rangle
    = -D\left\|\sum_{t=1}^T \nabla F(\vecw_t^k) \right\| + \regret_T(\vecu^k) + \sum_{t=1}^T \langle \nabla F(\vecw_t^k)-\tilde\vecg_t^k, \Delta_t^k-\vecu^k \rangle.
\end{equation*}
Upon rearranging the inequality and applying $F(\vecx_{T+1}^K)-F(\vecx_1^1)\le F^*$, we have
\begin{align*}
    \sum_{k=1}^K D\E\left\|\sum_{t=1}^T\nabla F(\vecw_t^k)\right\|
    &\le F^* + \sum_{k=1}^K \E\regret_T(\vecu^k) + \E\sum_{k=1}^K\sum_{t=1}^T \langle \nabla F(\vecw_t^k)-\tilde\vecg_t^k, \Delta_t^k-\vecu_t^k \rangle.
\end{align*}
Finally, note that $\|\vecw_t^k-\overline w^k\| \le DT = \delta$, so $\|\nabla F(\overline w^k)\|_\delta \le \|\frac{1}{T}\sum_{t=1}^T \nabla F(\vecw_t^k)\|$ by definition of $\|\cdot\|_\delta$. Moreover, $\E\|\nabla F(\overline\vecw)\|_\delta = \E[\frac{1}{K}\sum_{i=1}^K\|\nabla F(\overline w^k)\|_\delta]$. Therefore, dividing both sides of the inequality by $DKT$ concludes the proof.
\end{proof}

% \VarianceLemma*

\begin{proof}
First, we recall a martingale concentration bound. Let $\vecv_1,\ldots,\vecv_n$ be a sequence of random vectors and let $\calF_i=\sigma(\vecv_{1:i})$ be the $\sigma$-algebra generated by $\vecv_{1:i}$. If $\E[\vecv_i|\calF_{i-1}] = 0$, then for all cross terms $i\le j$, $v_i\in\calF_{j-1}$ and thus $\E\langle \vecv_i, \vecv_j\rangle = \E_{\calF_{j-1}}\langle \vecv_i, \E[\vecv_j|\calF_{j-1}]\rangle = 0$. Consequently,
\begin{equation*}
    \textstyle
    \E\|\sum_{i=1}^n \vecv_i\|^2 = \sum_{i=1}^n \E\|\vecv_i\|^2.
\end{equation*}
In our case, note that $\nabla\hat F(\vecw_t^k) = \nabla\hat F(\vecw_1^k) + \sum_{i=2}^t \nabla\hat F(\vecw_t^k) - \nabla\hat F(\vecw_{t-1}^k)$ and recall that $\vecg_t^k = \vecg_1^k + \sum_{i=2}^t \vecd_i$. Therefore, by the concentration inequality and Lemma \ref{lem:DP-O2NC:grad-estimator} and \ref{lem:DP-O2NC:diff-estimator}, 
\begin{align*}
    \E\| \nabla \hat F(\vecw_t^k) - \vecg_t^k \|^2
    &\le \E\|\nabla \hat F(\vecw_1^k) - \vecg_1^k\|^2 + \sum_{i=2}^t \E\|\nabla\hat F(\vecw_i^k)-\nabla\hat F(\vecw_{i-1}^k) - \vecd_i^k\|^2 \\
    &\le \frac{16dL^2}{B_1} + \sum_{i=2}^t \frac{16dL^2}{B_2\delta^2} \E\|\vecw_i^k-\vecw_{i-1}^k\|^2.
\end{align*}
We conclude the proof by Remark \ref{rmk:DP-O2NC:w-distance} that $\|\vecw_i^k-\vecw_{i-1}^k\| \le 2D \le \frac{2\delta}{T}$.
\end{proof}
