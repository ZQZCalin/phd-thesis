\section{Proofs of Theorem \ref{thm:Exp-O2NC:adam}}

First, we provide a technical proposition that is widely used in AdaGrad-style analysis.

\begin{proposition}
\label{prop:Exp-O2NC:tech-adagrad}
% Let $a_0\ge 0$ and $f:[0,\infty)\to[0,\infty)$ be non-increasing. Moreover, denote $a_{0:t} = \sum_{i=0}^t a_i$. Then
% \begin{equation*}
%     \sum_{t=1}^T a_tf(a_{0:t}) \le \int_{a_0}^{a_{0:T}} f(x)\, dx.
% \end{equation*}
For any $a_0\ge 0$,
\begin{equation*}
    \sum_{t=1}^T \frac{a_t}{\sqrt{\epsilon_0+\sum_{i=1}^t a_i}} 
    % \le 2\sqrt{\epsilon_0 + \sum_{t=1}^T a_t} - 2\sqrt{\epsilon_0} 
    \le 2\sqrt{\sum_{t=1}^T a_t}.
\end{equation*}
\end{proposition}

\begin{proof}
Let $a_{0:t}$ denote $\sum_{i=0}^t a_i$ and let $f:[0,\infty)\to [0,\infty)$ be a non-increasing function. Then $f(a_{0:t}) \le f(x)$ for all $x\in [a_{0:t-1},a_{0:t}]$. Consequently,
\begin{align*}
    a_t f(a_{0:t})
    &= \int_{a_{0:t-1}}^{a_{0:t}} f(a_{0:t}) \, dx
    \le \int_{a_{0:t-1}}^{a_{0:t}} f(x) \, dx.
\end{align*}
Upon summing over $t$ from $1$ to $T$, we have
\begin{align*}
    \sum_{t=1}^T a_t f(a_{0:t}) \le \int_{a_0}^{a_{0:T}} f(x) \,dx.
\end{align*}
We then conclude the proof by substituting $f(x) = \frac{1}{\sqrt{x}}$ and $\int_{a_0}^{a_{0:T}} f(x)\, dx = 2\sqrt{a_{0:T}}-2\sqrt{a_0} \le 2\sqrt{a_{1:T}}$.
\end{proof}


To prove Theorem \ref{thm:Exp-O2NC:adam}, we need a slightly modified version of Theorem \ref{thm:Exp-O2NC:o2nc-formal}, which we prove below. Note that the assumptions and definitions follows exactly the same as in Theorem \ref{thm:Exp-O2NC:o2nc-formal}, and the proof is also identical except for one specific equation.

\begin{theorem}
\label{thm:Exp-O2NC:o2nc-adaptive}
Suppose $F$ is $G$-Lipschitz, $F(\vecx_0)-\inf F(\vecx) \le F^*$, and the stochastic gradients satisfy $\E[\nabla f(\vecx,z)\,|\,\vecx] = \nabla F(\vecx)$ and $\E\|\nabla F(\vecx)-\nabla f(\vecx,z)\|^2\le \sigma^2$ for all $\vecx,z$. Define the comparator $\vecu_n$ and the regret $\regret_n(\vecu)$ of the regularized losses $\ell_t$ as follows:
\begin{align*}
    & \vecu_n = -D \cdot \frac{\sum_{t=1}^n \beta^{n-t}\nabla F(\vecx_t)}{\|\sum_{t=1}^n \beta^{n-t}\nabla F(\vecx_t)\|},
    & \regret_n(\vecu) 
    % = \sum_{t=1}^n \ell_t(\Delta_t)
    = \sum_{t=1}^n \langle \beta^{-t}\vecg_t, \Delta_t - \vecu \rangle + \calR_t(\Delta_t) - \calR_t(\vecu).
\end{align*}
Also define the regularizor as $\calR_t(\vecw) = \frac{\mu_t}{2}\|\vecw\|^2$ where $\mu_t = \mu\beta^{-t}$, $\mu=\frac{24cD}{\alpha^2}$ and $\alpha=1-\beta$. Furthermore, denote $S_N^2 = \frac{1}{N}\sum_{t=1}^N \E\|\nabla F(\vecx_t) - \vecg_t\|^2$. Then
\begin{align*}
    \E \|\nabla F(\overline\vecx)\|_c
    &\le \frac{F^*}{DN} + \frac{2G+\sigma}{\alpha N} + S_N\sqrt{\alpha} + \frac{12c D^2}{\alpha^2} 
    + \frac{1}{DN}\left( \beta^{N+1}\E\regret_N(\vecu_N) + \alpha \sum_{n=1}^N \beta^n \E\regret_n(\vecu_n). \right).
\end{align*}
\end{theorem}

\begin{proof}
The proof completely mirrors the proof of Theorem \ref{thm:Exp-O2NC:o2nc-formal}, while we modify the bound in \eqref{eq:Exp-O2NC:exp-2b}. Recall that \eqref{eq:Exp-O2NC:exp-2b} bounds the variance term, which is copied as follows:
\begin{align*}
    \E\sum_{n=1}^N\sum_{t=1}^n \beta^{n-t}(1-\beta) \langle \nabla F(\vecx_t) - \vecg_t, -\vecu_n\rangle 
    &\le \sum_{n=1}^N (1-\beta) \cdot \frac{\sigma D}{\sqrt{1-\beta^2}}
    \le \sigma DN\sqrt{1-\beta}.
\end{align*}
Instead, we provide an adaptive bound involving $S_N$ defined in Theorem \ref{thm:Exp-O2NC:o2nc-adaptive}. By Cauchy-Schwarz inequality and martingale identity (recall that $\E[\nabla F(\vecx_t)-\vecg_t|\calF_t] = 0$ where $\calF_t$ denotes the $\sigma$-algebra generated by $\vecx_t$),
\begin{align*}
    \E\sum_{t=1}^n \beta^{n-t}\langle \nabla F(\vecx_t)-\vecg_t,-\vecu_n\rangle 
    &\le D \sqrt{\E \left\| \sum_{t=1}^n \beta^{n-t}(\nabla F(\vecx_t)-\vecg_t) \right\|^2} \tag{Cauchy-Schwarz}\\
    &= D\sqrt{\sum_{t=1}^n \beta^{2(n-t)} \E\|\nabla F(\vecx_t)-\vecg_t\|^2}. \tag{martingale identity}
\end{align*}
Next, since $\sqrt{\cdot}$ is concave, $\sum_{n=1}^N\sqrt{a_n} \le \sqrt{N\sum_{n=1}^N a_n}$. Consequently, by change of summation and $\alpha=1-\beta$,
\begin{align}
    \E\sum_{n=1}^N\sum_{t=1}^n \beta^{n-t}(1-\beta) \langle \nabla F(\vecx_t) - \vecg_t, -\vecu_n\rangle 
    &\le (1-\beta) \sum_{n=1}^N D\sqrt{\sum_{t=1}^n \beta^{2(n-t)} \E\|\nabla F(\vecx_t)-\vecg_t\|^2} \notag\\
    &\le \alpha D\sqrt{N} \sqrt{\sum_{n=1}^N\sum_{t=1}^n \beta^{2(n-t)} \E\|\nabla F(\vecx_t)-\vecg_t\|^2} \notag\\
    &= \alpha D\sqrt{N} \sqrt{\sum_{t=1}^N \left(\sum_{n=t}^N \beta^{2(n-t)}\right) \E\|\nabla F(\vecx_t)-\vecg_t\|^2} \notag\\
    &\le \alpha DN \sqrt{ \frac{1}{N}\sum_{t=1}^N \frac{1}{1-\beta}\E\|\nabla F(\vecx_t)-\vecg_t\|^2}
    = S_NDN\sqrt{\alpha}.
    \label{eq:Exp-O2NC:exp-2b-adaptive}
\end{align}
Here the last inequality uses $\sum_{n=t}^N \beta^{2(n-t)} = \frac{1-\beta^{2(N-t+1)}}{1-\beta^2} \le \frac{1}{1-\beta}$ since $\beta<1$ and substitutes the definition of $S_N$. Consequently, upon substituting \eqref{eq:Exp-O2NC:exp-2b} with \eqref{eq:Exp-O2NC:exp-2b-adaptive} in the proof of Theorem \ref{thm:Exp-O2NC:o2nc-formal} while keeping the rest the same, we then conclude the proof of this modified theorem.
\end{proof}


\Adam*

\begin{proof}[Proof of Theorem \ref{thm:Exp-O2NC:adam}]
% The proof follows a similar structure to the proof of Theorem \ref{thm:sgdm}. Let $D=\frac{F^*}{(G+\sigma)\sqrt{\alpha}N}, \mu=\frac{24cD}{\alpha^2}, \eta=D$ such that $\mu,\eta$ also satisfy $\mu=\frac{24F^*c}{(G+\sigma)\alpha^{5/2}N}, \eta=\frac{F^*}{(G+\sigma)\sqrt{\alpha}N}$ as defined in the theorem. Also 
Note that the adaptive learning rate $\eta_t$ is non-increasing. Hence, both Theorem \ref{thm:Exp-O2NC:o2nc-adaptive} and Theorem \ref{thm:Exp-O2NC:OGD-formal} hold.
Since $\|\vecg_t\|\le G_f$, by Proposition \ref{prop:Exp-O2NC:tech-adagrad},
\begin{align*}
    \sum_{t=1}^n \eta_t\|\beta^{-t}\vecg_t\|^2
    &= \sum_{t=1}^n \frac{\eta\|\beta^{-t}\vecg_t\|^2}{\sqrt{\epsilon_0+\sum_{i=1}^{t-1}\|\beta^{-i}\vecg_i\|^2 + (\beta^{-t}G_f)^2}} \\
    &\le \sum_{t=1}^n \frac{\eta\|\beta^{-t}\vecg_t\|^2}{\sqrt{\epsilon_0+\sum_{i=1}^{t}\|\beta^{-i}\vecg_i\|^2}} 
    \le 2\eta \sqrt{\sum_{t=1}^n \|\beta^{-t}\vecg_t\|^2}.
\end{align*}
Therefore, by Theorem \ref{thm:Exp-O2NC:OGD-formal},
\begin{align*}
    \E\regret_n(\vecu_n)
    &\le \left(\frac{2}{\eta_{n+1}}+\frac{\mu_{n+1}}{2}\right)D^2 + \frac{1}{2}\sum_{t=1}^n\eta_t\E\|\beta^{-t}\vecg_t\|^2 \\
    &\le \left(\frac{2D^2}{\eta} + \eta\right) \sqrt{\sum_{t=1}^n\E\|\beta^{-t}\vecg_t\|^2+(\beta^{-(n+1)}G_f)^2} + \frac{\mu_{n+1}}{2}D^2
    \intertext{Upon substituting $\eta=D$, $\mu_t=\frac{24cD}{\alpha^2}\beta^{-t}$ and $\|\vecg_t\|\le G_f$, we have}
    &\le 3D\cdot \left( \beta^{-(n+1)}G_f + \sqrt{\sum_{t=1}^n\E\|\beta^{-t}\vecg_t\|^2} \right) + \frac{12cD^3 \beta^{-(n+1)}}{\alpha^2}.
\end{align*}
Consequently, with $\alpha \le \frac{1}{2}$ so that $\beta^{-1}\le 2$,
\begin{align}
    &\frac{1}{DN}\E\left( \beta^{N+1}\regret_N(\vecu_N) + \alpha\sum_{n=1}^N \beta^n\regret_n(\vecu_n) \right) \notag\\
    &\le \frac{1+2\alpha N}{DN} \left( 3DG_f + \frac{12cD^3}{\alpha^2} \right) + \frac{3D}{DN}\left( \sqrt{\sum_{t=1}^N \E\|\beta^{N-t}\vecg_t\|^2} + \alpha\sum_{n=1}^N\sqrt{\sum_{t=1}^n \E\|\beta^{n-t}\vecg_t\|^2} \right).
    \label{eq:Exp-O2NC:adagrad-1a}
    % &\lesssim \frac{G_f}{N} + \frac{cD^2}{\alpha N} + G_f\sqrt{\alpha} + \frac{cD^2}{\alpha}.
\end{align}
Since $\beta<1$, with the assumption that $\alpha N\ge 1$, we have
\begin{align*}
    \sqrt{\sum_{t=1}^N \E\|\beta^{N-t}\vecg_t\|^2}
    &\le \sqrt{N}\sqrt{\frac{1}{N}\sum_{t=1}^N \E\|\vecg_t\|^2}
    = \sqrt{N}V_N \le NV_N\sqrt{\alpha}.
\end{align*}
Next, since $\sqrt{\cdot}$ is concave, $\sum_{n=1}^N\sqrt{a_n} \le \sqrt{N\sum_{n=1}^N a_n}$. Hence,
\begin{align*}
    \alpha\sum_{n=1}^N\sqrt{\sum_{t=1}^n\|\beta^{n-t}\vecg_t\|^2}
    &\le \alpha\sqrt{N}\sqrt{\sum_{n=1}^N\sum_{t=1}^n\beta^{2(n-t)}\|\vecg_t\|^2} \\
    &= \alpha\sqrt{N} \sqrt{\sum_{t=1}^N\left(\sum_{n=t}^N \beta^{2n-t}\right) \|\vecg_t\|^2} \tag{change of summation}\\
    &\le \alpha N\sqrt{ \frac{1}{N}\sum_{t=1}^N \frac{1}{1-\beta} \|\vecg_t\|^2} = NV_N\sqrt{\alpha}.
\end{align*}
Here the last line uses $\sum_{n=t}^N \beta^{2n} = \frac{\beta^{2t}-\beta^{2(n+1)}}{1-\beta^2} \le \frac{\beta^{2t}}{1-\beta}$ since $\beta<1$ and substitutes the definition of $V_N$. Therefore, upon substituting back into \eqref{eq:Exp-O2NC:adagrad-1a} with $\alpha N \ge 1$, we have
\begin{align*}
    \frac{1}{DN}\left( \beta^{N+1}\regret_N(\vecu_N) + \alpha\sum_{n=1}^N \beta^n\regret_n(\vecu_n) \right) 
    % &\lesssim \frac{\alpha}{D}\left(D G_f + \frac{cD^3}{\alpha^2}\right) + \frac{1}{N} \cdot NV_N\sqrt{\alpha} \\
    &\lesssim G_f\alpha + \frac{cD^2}{\alpha} + V_N\sqrt{\alpha}.
\end{align*}
Upon substituting this into the convergence guarantee in Theorem \ref{thm:Exp-O2NC:o2nc-adaptive}, we have
\begin{align*}
    \E \|\nabla F(\overline\vecx)\|_c
    &\le \frac{F^*}{DN} + \frac{2G+\sigma}{\alpha N} + S_N\sqrt{\alpha} + \frac{12c D^2}{\alpha^2} 
    + \frac{1}{DN}\left( \beta^{N+1}\E\regret_N(\vecu_N) + \alpha \sum_{n=1}^N \beta^n \E\regret_n(\vecu_n) \right) \\
    &\lesssim \frac{F^*}{DN} + \frac{G+\sigma}{\alpha N} + (S_N+V_N)\sqrt{\alpha} + \frac{cD^2}{\alpha^2} + G_f\alpha
    \intertext{
    With $D=(\frac{F^*\alpha^2}{cN})^{1/3}$
    % With $D=\frac{F^*}{(G+\sigma)\sqrt{\alpha}N}$ 
    and $\alpha=\max\{N^{-2/3}, \frac{(F^*)^{4/7}c^{2/7}}{N^{4/7}}\}$ (this automatically satisfies $\alpha N \ge 1$), we have}
    &\lesssim \frac{G+\sigma}{\alpha N} + (S_N+V_N)\sqrt{\alpha} + 
    \frac{(F^*)^{2/3}c^{1/3}}{\alpha^{2/3}N^{2/3}} + G_f\alpha \\
    % \frac{(F^*)^2c}{\alpha^3N^2}
    &\lesssim \frac{G+\sigma}{N^{1/3}} + \frac{(S_N+V_N+1)(F^*)^{2/7}c^{1/7}}{N^{2/7}} + \frac{G_f}{N^{2/3}} +\frac{G_f(F^*)^{4/7}c^{2/7}}{N^{4/7}}.
    \qedhere
\end{align*}
\end{proof}
