        \Statex \textbf{Input:} Index $t\in[T]$, global variables $\sigma_1,\ldots,\sigma_T$.
        % \If {$r = 1$}

        \State Return $\sum_{(\cdot,i)\in \textsc{Node}(t)} \xi_i$.
