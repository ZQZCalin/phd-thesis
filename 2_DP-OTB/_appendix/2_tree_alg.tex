\section{Proofs for RDP (Section \ref{sec:DP-OTB_OTB})}
\label{app:DP-OTB:tree-aggregation-RDP}

\ThmDpOtbPrivacy*

\begin{proof}
By Theorem \ref{thm:tree-mechanism:privacy}, the composition of $\tilde\vecg_t=\vecg_t + \tree(t)$ is $(\alpha,\alpha\rho^2/2)$-RDP, where
\begin{equation*}
    \rho \le \sqrt{2\log_2(2t)}\cdot \max_{i\in[t]} \frac{\max_{j\le i} s_j}{\sigma_i},
\end{equation*}
and $s_i$ is the sensitivity of the gradient difference $\vecd_i$, which by \eqref{eq:app:DP-OTB:grad-diff-sens} is bounded by 
\begin{equation*}
    \|\vecd_t\|_\ast \le (k+1)(G+DH)t^{k-1}.
\end{equation*}
Consequently, we can verify that the value in \eqref{eq:DP-OTB:RDP-noise} indeed achieves the privacy level $\rho$.

Finally, we conclude the proof by noting that Algorithm \ref{alg:DP-OTB:main} is a post-processing of the composition of $\tilde\vecg_t$'s.
\end{proof}
