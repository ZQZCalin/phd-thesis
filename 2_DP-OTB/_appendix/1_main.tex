\LemAnytimeOTB*

\begin{proof}
Since $f$ is convex,
\begin{align*}
    \sum_{t=1}^T \weight_t ( f(x_t) - f(u) )
    &\leq \sum_{t=1}^T \weight_t \langle \nabla f(x_t), x_t-w_t+w_t-u \rangle \\
    &= \sum_{t=1}^T \langle\nabla f(x_t), \weight_t(x_t-w_t)\rangle + \langle \weight_t\nabla f(x_t)-g_t+g_t, w_t-u\rangle.
\end{align*}
By construction of $x_t$, it holds that $\weight_t(x_t-w_t) = \weight_{1:t-1}(x_{t-1}-x_t)$. By convexity,
\begin{align*}
    \langle \nabla f(x_t), \weight_{1:t-1}(x_{t-1}-x_t) \rangle \leq \weight_{1:t-1} \left( f(x_{t-1}) - f(x_t) \right).
\end{align*}
Next, we move $\sum \weight_tf(x_t)$ to the right, giving:
\begin{align*}
    -\weight_{1:T} f(u) 
    &\leq \sum_{t=1}^T \left( \weight_{1:t-1}f(x_{t-1}) - \weight_{1:t}f(x_t) + \langle \weight_t\nabla f(x_t)-g_t+g_t,w_t-u\rangle \right) \\
    % &= -\weight_{1:T}\nabla f(x_T) + \sum_{t=1}^T \langle\weight_t\nabla f(x_t)-g_t+g_t, w_t-x^*\rangle \\
    &= -\weight_{1:T}f(x_T) + \regret_T(u) + \sum_{t=1}^T \langle \weight_t\nabla f(x_T) - g_t,w_t-u\rangle,
\end{align*}
where the equality follows from (i) the telescopic sum of $\weight_{1:t-1}\nabla f(x_{t-1})+\weight_{1:t}\nabla f(x_t)$ and (ii) the definition of regret that $\regret_T(u)=\sum_{t=1}^T \langle g_t,w_t-u\rangle$.
\end{proof}

In this paper, we will always set $\weight_t=t^k$ for some $k\geq 1$. The following proposition gives relevant bounds for $\weight_t-\weight_{t-1}$ and $\weight_t^2/\weight_{1:t}$.

\begin{proposition}
\label{prop:DP-OTB:alphas}
If $\weight_t=t^k$, then (i) $\weight_t-\weight_{t-1} \leq kt^{k-1}$ and (ii) $\weight_t^2/\weight_{1:t} \leq (k+1)t^{k-1}$.
% \begin{align*}
%     (1) \quad & \weight_t-\weight_{t-1} \leq kt^{k-1}, \\
%     (2) \quad & \weight_t^2/\weight_{1:t} \leq (k+1)t^{k-1}.
% \end{align*}
\end{proposition}

\begin{proof}
For the first part, by mean value theorem, there exists some $\tau\in [t-1,t]$ such that
\begin{align*}
    \weight_t - \weight_{t-1} = t^k - (t-1)^k
    = k\tau^{k-1} \leq kt^{k-1}.
\end{align*}
For the second part, for any increasing function $f$, it holds that $\sum_{i=1}^t f(i) \geq \int_0^t f(x) \, dx$, so
\begin{align*}
    \weight_{1:t} = \sum_{i=1}^t i^k \geq \int_0^t x^k \, dx = \frac{t^{k+1}}{k+1}.
\end{align*}
Hence, $\weight_t^2/\weight_{1:t} \leq (k+1)t^{k-1}$.
\end{proof}

\begin{lemma}[Lemma \ref{lem:DP-OTB:variance-expectation-bound}, detailed version]
\label{lem:DP-OTB:variance-expectation-bound:detailed}
Suppose $\|\cdot\|^2$ is $\lambda$-strongly convex, $\calX$ is bounded by $D$, and $F$ is $G$-Lipschitz and $H$-smooth. If we set $\weight_t=t^k$, then
\begin{equation}
    \|\vecd_t\|_* \leq (k+1) (G+H\|w_t-x_{t-1}\|) t^{k-1}.
\end{equation}
If we further assume Assumption \ref{as:DP-OTB:sigma-G} and \ref{as:DP-OTB:sigma-H},
then $\weight_t\nabla f(x_t)-g_t = \sum_{i=1}^tX_i$ such that:
\begin{align*}
(i)\ & X_i = [\weight_i\nabla f(x_i)-\weight_{i-1}\nabla f(x_{i-1})] - [\weight_i\nabla\ell(x_i,z_i)-\weight_{i-1}\nabla\ell(x_{i-1},z_i)], \\
(ii)\ & \E[X_i|z_{1:i-1}]=0, \quad \text{and} \quad
(iii)\ \E[\|X_i\|_*^2] 
\leq 2(k+1)^2(\sigma_G^2+D^2\sigma_H^2)i^{2k-2}.
\end{align*}
Consequently,
\begin{equation}
    \E[\|\weight_t\nabla f(x_t)-g_t\|_*^2] \leq 4(k+1)^2(\sigma_G^2+D^2\sigma_H^2)t^{2k-1}/\lambda.
\end{equation}
\end{lemma}

\begin{proof}
For the first part, note that
\begin{align*}
    \vecd_t &= \weight_t\nabla\ell(x_t,z_t)-\weight_{t-1}\nabla\ell(x_{t-1},z_t) \\
    &= (\weight_{t}-\weight_{t-1}) \nabla\ell(x_{t-1},z_t) + \weight_t(\nabla\ell(x_t,z_t)-\nabla\ell(x_{t-1},z_t)).
\end{align*}
Since $\ell$ is $G$-Lipschitz and $H$-smooth, $\|\nabla\ell(x_{t-1},z_t)\|_*\leq G$ and
\begin{align*}
    \|\nabla\ell(x_t,z_t)-\nabla\ell(x_{t-1},z_t)\|_* 
    &\leq H\|x_t-x_{t-1}\| \leq (\weight_t/\weight_{1:t})H\|w_t-x_{t-1}\|,
\end{align*}
where the second inequality follows from the definition of $x_t$ that $\weight_{1:t}(x_t-x_{t-1})=\weight_t(w_t-x_{t-1})$. The first result then follows from Proposition \ref{prop:DP-OTB:alphas}.

For the second part, by telescopic sum $\weight_t\nabla f(x_t) = \sum_{i=1}^t \weight_i\nabla f(x_i) - \weight_{i-1}\nabla f(x_{i-1})$, and recall that $g_t=\sum_{i=1}^t \vecd_i$. Therefore,
\begin{align*}
    \weight_t\nabla f(x_t)-g_t 
    &= \sum_{i=1}^t [\weight_i\nabla f(x_i) - \weight_{i-1}\nabla f(x_{i-1})] - [\weight_i\nabla\ell(x_i,z_i)-\weight_{i-1}\nabla\ell(x_{i-1},z_i)].
\end{align*}
We denote each summand by $X_i$, and we can check $X_i$ satisfies condition (ii) and (iii). First, since we assume $\nabla f(w) = \E_z[\nabla\ell(w,z)]$ for all $w$, it holds that $\E[X_i|z_{1:i-1}]=0$. Second, we decompose $X_i$ in the same way as the first part: 
\begin{align*}
    \E[\|X_i\|_*^2]
    &= \E[ \|(\weight_i-\weight_{i-1})[\nabla(f(x_{i-1})-\nabla\ell(x_{i-1},z_i)] \\
    &\quad\ + \weight_{i}( [\nabla f(x_i)-\nabla\ell(x_i,z_i)] - [\nabla f(x_{i-1})-\nabla\ell(x_{i-1},z_i)] ) \|_*^2 ] \\
    \intertext{Recall the assumption of $\sigma_G^2$ and $\sigma_H^2$ and that $\|x_i-x_{i-1}\| \leq \weight_i/\weight_{1:i}\|w_i-x_{i-1}\|$.}
    &\leq 2(\weight_i-\weight_{i-1})^2\sigma_G^2 + 2\weight_i^2 \sigma_H^2\|x_i-x_{i-1}\|^2 \\
    &\leq 2(\weight_i-\weight_{i-1})^2\sigma_G^2 + 2(\weight_i^2/\weight_{1:i})^2 \sigma_H^2\|w_i-x_{i-1}\|^2 \\
    &\leq 2(k+1)^2(\sigma_G^2 + D^2\sigma_H^2)i^{2k-2}.
\end{align*}
The last inequality follows from the assumption that $\|w_i-x_{i-1}\|\leq D$ and Proposition \ref{prop:DP-OTB:alphas}.

The last part of the theorem is a direct result from Theorem \ref{thm:concentration:martingale-difference}:
\begin{align*}
    \E[ \| \nabla f(x_t)-g_t \|_*^2]
    \leq \frac{2}{\lambda} \sum_{i=1}^t \E[\|X_i\|_*^2] 
    % &\leq \frac{2}{\lambda} \sum_{i=1}^t 2k^2i^{2k-2}\sigma_G^2 + 2(k+1)^2i^{2k-2}D^2\sigma_H^2 \\
    % &\leq \frac{2}{\lambda} \sum_{i=1}^t 2(k+1)^2(\sigma_G^2+D^2\sigma_H^2) i^{2k-2} \\
    % &\leq 4(k^2\sigma_G^2+(k+1)^2D^2\sigma_H^2)t^{2k-1}/\lambda.
    &\leq \frac{4(k+1)^2}{\lambda}(\sigma_G^2+D^2\sigma_H^2)t^{2k-1}.
\end{align*}
\end{proof}
