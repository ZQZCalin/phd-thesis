\begin{proof}
We start with \eqref{eq:DP-OTB:basic-decomposition-2}:
\begin{align*}
    \alpha_{1:T}(\L(x_T)-\L(x^*))
    \leq R_T(x^*) + \sum_{t=1}^T\langle \alpha_t\nabla\L(x_t)-g_t-\gamma_t,w_t-x^*\rangle.
\end{align*}
By Fenchel-Young's inequality and triangle inequality,
\begin{align*}
    \sum_{t=1}^T\langle \alpha_t\nabla\L(x_t)-g_t-\gamma_t,w_t-x^*\rangle 
    \leq \sum_{t=1}^T \left(\|\alpha_t\nabla\L(x_t)-g_t\|_2+\|\gamma_t\|_2\right) (\|w_t-x^*\|_2).
\end{align*}

\textbf{Step 1.}
By Lemma \ref{lem:DP-OTB:nSG-concentration} and \ref{lem:DP-OTB:X-R-nSG}, for each $t$, with probability $1-\delta_t$,
\begin{align*}
    \|\alpha_t\nabla\L(x_t)-g_t\|_2 
    \leq C\sqrt{\sum_{i=1}^t \sigma_{X_i}^2 \log\frac{2d}{\delta_t}}.
\end{align*}
By Proposition \ref{prop:DP-OTB:alphas} that $\alpha_t-\alpha_{t-1}\leq kt^{k-1}$ and $\alpha_t^2/\alpha_{1:t}\leq (k+1)t^{k-1}$,
\begin{align*}
    \sigma_{X_i} \leq 2C(k+1)i^{k-1}(G+H\|w_i-x_{i-1}\|_2).
\end{align*}
Define $\delta_t=\frac{\delta}{(\pi^2/6)t^2}$, then $\sum_{t=1}^T \delta_t\leq \delta$. Therefore, by union bound, with probability at least $1-\delta$,
\begin{align*}
    &\sum_{t=1}^T \|\alpha_t\nabla\L(x_t)-g_t\|_2\|w_t-x^*\|_2 \\
    \leq &\sum_{t=1}^T C\sqrt{\log\frac{2d}{\delta_t}\sum_{i=1}^t [2C(k+1)i^{k-1}(G+H\|w_i-x_{i-1}\|_2)]^2} \|w_t-x^*\|_2
    \intertext{We use the identity $(a+b)^2\leq 2a^2+2b^2$ and $\sqrt{a+b}\leq \sqrt{a}+\sqrt{b}$:}
    \leq & 2(k+1)C^2\sum_{t=1}^T \sqrt{\log\frac{2d}{\delta_t}}\left(\sqrt{\sum_{i=1}^t 2G^2i^{2(k-1)}} + \sqrt{\sum_{i=1}^t 2H^2\|w_i-x_{i-1}\|_2^2i^{2(k-1)}}\right) \|w_t-x^*\|_2
\end{align*}
\textbf{1.1.} We bound these two sums separately. For the first sum,
\begin{align*}
    \sqrt{\sum_{i=1}^t 2G^2i^{2(k-1)}} \leq \sqrt{2}Gt^{k-\frac{1}{2}}.
\end{align*}
Moreover, by Young's inequality that $ab\leq \frac{1}{2\lambda}a^2+\frac{\lambda}{2}b^2$, $\|w_t-x^*\|_2 \leq \frac{1}{2}(1+\|w_t-x^*\|^2)$. Hence,
\begin{align*}
    \sum_{t=1}^T \sqrt{\log\frac{2d}{\delta_t}\sum_{i=1}^t 2G^2i^{2(k-1)}}\|w_t-x^*\|_2 
    \leq & G\sum_{t=1}^T \sqrt{2\log \frac{\pi^2dt^2}{3\delta}}t^{k-\frac{1}{2}}(1+\|w_t-x^*\|_2^2)/2.
\end{align*}
\textbf{1.2.}
For the second sum, we apply Young's inequality for each $t$ and apply change of summation:
\begin{restatable}{lemma}{LemmaChangeSum}
\label{lem:DP-OTB:change-sum}
For any sequence $a_i,b_j,c_k$,
\begin{align*}
    \sum_{i=1}^N a_i \sum_{j=1}^i b_j = \sum_{i=1}^N b_i \sum_{j=i}^N a_j,
    \quad \textit{and} \quad 
    \sum_{i=1}^N a_i \sum_{j=1}^i b_j \sum_{k=1}^j c_k = \sum_{i=1}^N c_i \sum_{j=i}^N a_j \sum_{k=i}^j b_k.
\end{align*}
\end{restatable}
The second sum can be bounded by:
\begin{align*}
    &\sum_{t=1}^T\sqrt{\log\frac{2d}{\delta_t}\sum_{i=1}^t2H^2\|w_i-x_{i-1}\|_2^2i^{2(k-1)}}\|w_t-x^*\|_2 \\
    \leq & H\sum_{t=1}^T \sqrt{\log\frac{2d}{\delta_t}} \left(\frac{1}{2\lambda_t}\sum_{i=1}^t (2\|w_i-x_{i-1}\|_2^2i^{2(k-1)}) + \frac{\lambda_t}{2}\|w_t-x^*\|_2^2\right)
    % line 1
    \intertext{We first bound $\|w_i-x_{i-1}\|_2^2 \leq 2\|w_i\|_2^2+2\|x_{i-1}\|_2^2$. Recall that $x_0=0$ and for $i\geq 2$, $x_{i-1}=\sum_{j=1}^{i-1} \frac{\alpha_j}{\alpha_{1:{i-1}}}w_j$, so $\|x_{i-1}\|_2^2\leq \sum_{j=1}^{i-1}\frac{\alpha_j}{\alpha_{1:i-1}}\|w_j\|_2^2$ by convexity. Hence. we further bound:}
    \leq & H\sum_{t=1}^T \sqrt{\log\frac{2d}{\delta_t}} \left(\frac{1}{ \lambda_t}\sum_{i=1}^t 2\|w_i\|_2^2i^{2(k-1)} + \frac{1}{\lambda_t}\sum_{i=2}^t 2i^{2(k-1)}\sum_{j=1}^{i-1} \frac{\alpha_j\|w_j\|_2^2}{\alpha_{1:i-1}} + \frac{\lambda_t}{2}\|w_t-x^*\|_2^2\right).
    % line 2
    % \intertext{Now we apply change of summation on the first two terms:}
    % \leq & H\sum_{t=1}^T \left( t^{2k-2}\|w_t\|^2\left(\sum_{i=t}^T\frac{\xi_i}{\lambda_i}\right) + \alpha_t\|w_t\|^2 \left(\sum_{i=t}^T \frac{\xi_i}{\lambda_i}\sum_{j=t}^i \frac{j^{2k-2}}{\alpha_{1:j}}\right) + \lambda_t(\|w_t\|^2+\|x^*\|^2) \right).
\end{align*}
\textbf{1.2.1.} Now we apply change of summation on the first two terms. For the first term,
\begin{align*}
    \sum_{t=1}^T \frac{1}{\lambda_t}\sqrt{\log \frac{2d}{\delta_t}} \sum_{i=1}^t 2\|w_i\|_2^2i^{2(k-1)} 
    = \sum_{t=1}^T\sum_{i=t}^T \frac{1}{\lambda_i}\sqrt{\log \frac{2d}{\delta_i}} 2\|w_t\|_2^2t^{2(k-1)}
\end{align*}
In general, we define $\lambda_t\propto t^{k-\frac{1}{2}}$. However, for simplicity, we consider the case when $k=3$, and define $\lambda_t = 14\sqrt{\log(2d/\delta_t)}t^{k-\frac{1}{2}}$. Then in the first sum, we then use the identity that $\sum_{i=t}^T f(i) \leq f(t)+\int_t^T f(x) dx$ for decreasing $f$:
\begin{align*}
    \sum_{i=t}^T \frac{14}{\lambda_i}\sqrt{\log \frac{2d}{\delta_i}}
    = \sum_{i=t}^T \frac{1}{i^{5/2}} \leq \frac{1}{t^{5/2}} + \int_t^\infty \frac{1}{x^{5/2}}\, dx = \frac{1}{t^{5/2}} + \frac{2}{3t^{3/2}} \leq \frac{5}{3t^{3/2}}.
\end{align*}

%     &\leq \frac{\sqrt{\log(\pi^2dt^2/3\delta)}}{t^{5/2}} + \int_t^\infty \frac{\sqrt{\log(\pi^2dx^2/3\delta)}}{x^{5/2}} \, dx

Therefore, the first term is bounded by:
\begin{align*}
    \sum_{t=1}^T\sum_{i=t}^T \frac{1}{\lambda_i}\sqrt{\log \frac{2d}{\delta_i}} 2\|w_t\|_2^2t^{2(k-1)}
    % \leq \sum_{t=1}^T 8\log\left(\frac{\pi^2dt^2}{3\delta}\right)t^{5/2}\|w_t\|_2^2.
    \leq \sum_{t=1}^T \frac{4}{14}t^{5/2}\|w_t\|_2^2.
\end{align*}
\textbf{1.2.2.} For the second term, we apply change of variable and then change of summation:
\begin{align*}
    \sum_{t=1}^T \frac{1}{\lambda_t}\sqrt{\log \frac{2d}{\delta_t}} \sum_{i=2}^t 2i^{2(k-1)}\sum_{j=1}^{i-1} \frac{\alpha_j\|w_j\|_2^2}{\alpha_{1:i-1}} 
    &\leq \sum_{t=1}^T \frac{1}{\lambda_t}\sqrt{\log \frac{d}{\delta_t}} \sum_{i=1}^{t} \frac{2(i+1)^{2(k-1)}}{\alpha_{1:i}}\sum_{j=1}^{i}2\alpha_j\|w_j\|_2^2\\
    &\leq \sum_{t=1}^T\sum_{i=t}^T\frac{1}{\lambda_i}\sqrt{\log\frac{2d}{\delta_i}}\sum_{j=t}^i\frac{(j+1)^{2(k-1)}}{\alpha_{1:j}} 2\alpha_t\|w_t\|_2^2.
\end{align*}
We then bound the inner summation with the same argument using the integral inequality. By Proposition \ref{prop:DP-OTB:alphas}, $\sum_{j=t}^i\frac{(j+1)^{2(k-1)}}{\alpha_{1:j}}\leq \sum_{j=t}^i\frac{4(j+1)^4}{j^4} \leq 64i$. Hence,

    &\leq \sum_{i=t}^T \frac{64}{i^{3/2}} \leq \frac{64}{t^{3/2}}+\int_t^\infty \frac{64}{x^{3/2}}\,dx = \frac{64}{t^{3/2}} + \frac{128}{t^{1/2}} \leq \frac{192}{t^{1/2}}.
\end{align*}
Consequently, the second term is bounded by:
\begin{align*}
    \sum_{t=1}^T\sum_{i=t}^T\frac{1}{\lambda_i}\sqrt{\log\frac{2d}{\delta_i}}\sum_{j=t}^i\frac{(j+1)^{2(k-1)}}{\alpha_{1:j}} 2\alpha_t\|w_t\|_2^2
    &\leq \sum_{t=1}^T \frac{192}{14}t^{5/2}\|w_t\|_2^2.
\end{align*}
In conclusion, \textbf{step 1.2.} gives the following bound:
\begin{align*}
    &\sum_{t=1}^T\sqrt{\log\frac{2d}{\delta_t}\sum_{i=1}^t2H^2\|w_i-x_{i-1}\|_2^2i^{2(k-1)}}\|w_t-x^*\|_2 \\
    \leq & H\sum_{t=1}^T 14t^{5/2}\|w_t\|_2^2 + 7\sqrt{\log(2d/\delta_t)}t^{5/2}\|w_t-x^*\|_2^2 \\
    \leq & H\sum_{t=1}^T \sqrt{\log \frac{\pi^2dt^2}{3\delta}}t^{5/2}(14\|w_t\|_2^2+7\|w_t-x^*\|_2^2).
\end{align*}
\textbf{Step 1.1.} and \textbf{1.2.} together implies the following result: with probability at least $1-\delta$,
\begin{align*}
    &\sum_{t=1}^T\|\alpha_t\nabla\L(x_t)-g_t\|_2\|w_t-x^*\|_2 \\
    % \leq & 2(k+1)C^2\sum_{t=1}^T \sqrt{\log\frac{\pi^2dt^2}{3\delta}}\left( Gt^{5/2}(1+\|w_t-x^*\|_2^2) + Ht^{5/2}(14\|w_t\|_2^2+7\|w_t-x^*\|_2^2) \right) \\
    \leq & 8C^2\sum_{t=1}^T \sqrt{\log\frac{\pi^2dt^2}{3\delta}}\left( Gt^{5/2} + (2G+28H)(\|w_t\|_2^2+\|x^*\|_2^2)t^{5/2} \right).
\end{align*}

\textbf{Step 2:} The proof for $\sum_{t=1}^T\|\gamma_t\|_2\|w_t-x^*\|_2$ follows similarly. By Lemma \ref{lem:DP-OTB:nSG-concentration} and \ref{lem:DP-OTB:normSG} and union bound, with probability $1-\delta$,
\begin{align*}
    &\sum_{t=1}^T\|\gamma_t\|_2\|w_t-x^*\|_2 \\
    \leq & \sum_{t=1}^TC\sqrt{ \log \frac{2d}{\delta_t}\sum_{i\in I_t}\sigma_{R_i}^2}\|w_t-x^*\|_2
    \intertext{We substitute $\sigma_{R_i}=C\sqrt{d}\sigma_\D\sigma_i$ and definition of $\sigma_i$ in \eqref{eq:DP-OTB:RDP-noise}:}
    \leq & \sum_{t=1}^T C\sqrt{ \log\frac{2d}{\delta_t}\sum_{i\in I_t} C^2d\sigma_\D^2\frac{2\pi^2(k+1)^2K}{\rho^2}(G+H\|w_i-x_{i-1}\|_2)^2\log_2^2(2i)i^{2(k-1)} }\|w_t-x^*\|_2 \\
    = & \frac{\sqrt{2dK}\pi(k+1)C^2\sigma_\D}{\rho} \sum_{t=1}^T \sqrt{\log\frac{2d}{\delta_t}\sum_{i\in I_t}(G+H\|w_i-x_{i-1}\|_2)^2\log_2^2(2i)i^{2(k-1)}}\|w_t-x^*\|_2.
\end{align*}
\highlight{At this point, we can definitely bound $\|w_i-x_{i-1}\|_2\leq D$ if we don't worry parameter-free for privacy. The following proof is an attempt to make it parameter-free.}
Then we apply the identity $\sqrt{a+b}\leq \sqrt{a}+\sqrt{b}$:
\begin{align*}
    &\sum_{t=1}^T \sqrt{\log\frac{2d}{\delta_t}\sum_{i\in I_t}(G+H\|w_i-x_{i-1}\|_2)^2\log_2^2(2i)i^{2(k-1)}}\|w_t-x^*\|_2 \\
    \leq & \sum_{t=1}^T \sqrt{2\log\frac{2d}{\delta_t}}\left(\sqrt{\sum_{i\in I_t}G^2\log_2^2(2i)i^{2k-2}}+\sqrt{\sum_{i\in I_t}H^2\|w_i-x_{i-1}\|_2^2\log_2^2(2i)i^{2k-2}}\right)\|w_t-x^*\|_2.
\end{align*}
\textbf{2.1.} Observe that $|I_t|\leq \log_2(2t)$, so
\begin{align*}
    \sqrt{\sum_{i\in I_t}G^2\log_2^2(2i)i^{2k-2}} \leq G\log_2^{3/2}(2t)t^{k-1}.
\end{align*}
\textbf{2.2.} By Young's inequality,
\begin{align*}
    &\sum_{t=1}^T \sqrt{2\log\frac{2d}{\delta_t}} \sqrt{\sum_{i\in I_t}\|w_i-x_{i-1}\|_2^2\log_2^2(2i)i^{2k-2}}\|w_t-x^*\|_2 \\
    \leq & \sum_{t=1}^T \sqrt{\log\frac{2d}{\delta_t}}\left( \frac{1}{\lambda_t}\sum_{i\in I_t}\log_2^2(2i)i^{2k-2}\left( 2\|w_i\|_2^2 + 2\|x_{i-1}\|_2^2 \right) + \frac{\lambda_t}{2}\|w_t-x^*\|_2^2 \right).
\end{align*}
We need a more general change of summation method. In general, let $A,B$ be two index sets and $\phi:A\to B$ a multi-valued mapping, and we define its pre-image as $\phi:b\mapsto \{a:b\in\phi(a)\}$. Then for any sequence $a_i,b_j$,
\begin{align*}
    \sum_{i\in A}\sum_{j\in \phi(i)} a_ib_j = \sum_{j\in B}\sum_{i\in \phi^{-1}(j)} a_ib_j.
\end{align*}
In our case, consider $A=B=[T]$ and $\phi:t\mapsto I_t$. Then $\phi^{-1}:t\mapsto J_t:=\{i\in[T]:t\in I_i\}$.
% \begin{align*}
%     \sum_{t=1}^T\sum_{i\in I_t} a_tb_i = \sum_{t=1}^T\sum_{i\in J_t} a_ib_t.
% \end{align*}
Now we can apply change of summation to the first two terms with the same arguments.

\textbf{2.2.1.} We define $\lambda_t = \log(2d/\delta_t)\log_2(2t)t^{k-1}$, and choose $k=3$ in this specific case. Then
\begin{align*}
    \sum_{t=1}^T \frac{1}{\lambda_t}\sqrt{\log\frac{2d}{\delta_t}} \sum_{i\in I_t} 2\|w_i\|_2^2\log_2^2(2i)i^{2k-2}
    &= \sum_{t=1}^T\sum_{i\in J_t} \frac{1}{\log_2(2i)i^2} 2\|w_t\|_2^2\log_2^2(2t)t^{4} \\
    &\leq \sum_{t=1}^T\sum_{i\in J_t} \frac{1}{i^2} 2\|w_t\|_2^2\log_2(2t)t^{4}
\end{align*}
The inequality follows from the observation that $i\geq t$ for all $i\in J_t$ and thus $\frac{1}{\log_2(2i)}\leq \frac{1}{\log_2(2t)}$.

\textbf{2.2.2.} Recall that $\|x_0\|_2=0$ and $\|x_{i-1}\|_2^2 \leq \sum_{j=1}^{i-1} \frac{\alpha_j\|w_j\|_2^2}{\alpha_{1:i-1}}$ and $1\in I_t$ iff $t=1$. Also recall that $\alpha_{1:t}\geq t^{k+1}/(k+1)$. Then
\begin{align*}
    &\sum_{t=1}^T \frac{1}{\lambda_t}\sqrt{\log\frac{2d}{\delta_t}}\sum_{i\in I_t} 2\log_2^2(2i)i^{2k-2}\|x_{i-1}\|_2^2 \\
    \leq & \sum_{t=2}^T \frac{1}{\log_2(2t)t^2} \sum_{i\in I_t} \frac{2(k+1)\log_2^2(2i)i^{2k-2}}{(i-1)^{k+1}} \sum_{j=1}^{i-1} j^k\|w_j\|_2^2. \tag{$*$}
\end{align*}
Next, we apply change of summation on $i$ and $j$. Consider $\psi:I_t\to [T]$ by $i\mapsto [i-1]$, then its pre-image is $\psi^{-1}:j\mapsto \{i\in I_t: j\in[i-1]\}=I_t\cap [j+1:T]$, where $[j+1:T]$ denotes the index set $\{j+1,j+2,\ldots,T\}$. Hence,
\begin{align*}
    \sum_{t=2}^T\sum_{i\in I_t}\sum_{j=1}^{i-1} a_tb_ic_j
    &= \sum_{j=1}^T\sum_{t=2}^T\sum_{\substack{i=j+1, i\in I_t}}^T  a_tb_ic_j
    \intertext{Observe that $I_t\subset[t]$, so $I_t\cap[j+1:T]=\varnothing$ for all $t\leq j$. This implies:}
    &= \sum_{j=1}^T\sum_{t=j+1}^T\sum_{\substack{i=j+1, i\in I_t}}^T  a_tb_ic_j
    \leq \sum_{j=1}^T\sum_{t=j+1}^T\sum_{i\in I_t} a_tb_ic_j.
\end{align*}
After re-indexing, we get:
\begin{align*}
    (*) &\leq \sum_{t=1}^T\sum_{i=t+1}^T \frac{1}{\log_2(2i)i^2} \sum_{j\in I_i} \frac{2(k+1)\log_2^2(2j)j^{2k-2}}{(j-1)^{k+1}} t^k\|w_t\|_2^2 
    \intertext{Recall that $|I_i|\leq \log_2(2i)$ and $j\leq i$ for all $j\in I_i$. Also, we can bound $j/(j-1)\leq 2$ because $1\not\in I_i$ for all $i\geq 2$ and thus $j\neq 1$.}
    &\leq \sum_{t=1}^T \sum_{i=t+1}^T \frac{512\log_2^2(2i)}{i^2} t^3\|w_t\|_2^2.
\end{align*}
By the integral bound,
\begin{align*}
    \sum_{i=t+1}^T \frac{\log_2^2(2i)}{i^2}
    &\leq \int_t^\infty \frac{\log_2^2(2x)}{x^2} \, dx
    \leq \frac{\log^2(2t) + 2\log(2t) + 2}{\log(2)^2 t}
    \leq \frac{\log_2^2(2t)+7\log_2(2t)+23}{t}.
\end{align*}
In conclusion, \textbf{2.2.} gives the following bound:

    &\sum_{t=1}^T \sqrt{2\log\frac{2d}{\delta_t}} \sqrt{\sum_{i\in I_t}\|w_i-x_{i-1}\|_2^2\log_2^2(2i)i^{2k-2}}\|w_t-x^*\|_2 \\
    \leq & \sum_{t=1}^T 
\end{align*}
